cmake_minimum_required(VERSION 3.20)
if(NOT DM_APP_SOURCE OR NOT DM_APP_NAME)
 message(FATAL_ERROR "Set DM_APP_SOURCE and DM_APP_NAME")
endif()
if(NOT DM_APP_LINUX)
 set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "toolchain")
endif()
project(${DM_APP_NAME} C)
option(DM_APP_NEWLIB "Initialize a private module newlib runtime (main-thread callbacks)" OFF)
set(DM_APP_LIBRARIES "" CACHE STRING "Extra native libraries, separated by semicolons")
set(DM_APP_INCLUDE_DIRS "" CACHE STRING "Extra include directories, separated by semicolons")
if(DM_APP_LINUX)
 add_library(${DM_APP_NAME} MODULE ${DM_APP_SOURCE})
 target_compile_definitions(${DM_APP_NAME} PRIVATE DM_PLUGIN_LINUX)
 target_link_libraries(${DM_APP_NAME} ${DM_APP_LIBRARIES})
 set_target_properties(${DM_APP_NAME} PROPERTIES PREFIX "" SUFFIX ".so")
else()
 include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)
 add_executable(${DM_APP_NAME} ${DM_APP_SOURCE})
 if(DM_APP_NEWLIB)
  target_compile_definitions(${DM_APP_NAME} PRIVATE DM_PLUGIN_NEWLIB)
  target_link_options(${DM_APP_NAME} PRIVATE -Wl,--wrap=__getreent)
  target_link_libraries(${DM_APP_NAME} -Wl,--start-group ${DM_APP_LIBRARIES} m c gcc SceNet_stub SceNetCtl_stub SceSysmodule_stub SceRtc_stub SceKernelThreadMgr_stub SceSysmem_stub SceLibKernel_stub SceIofilemgr_stub SceProcessmgr_stub SceKernelModulemgr_stub -Wl,--end-group)
 else()
  target_link_libraries(${DM_APP_NAME} ${DM_APP_LIBRARIES} SceLibc_stub SceLibKernel_stub gcc)
 endif()
 target_link_options(${DM_APP_NAME} PRIVATE -nostdlib -Wl,-q -Wl,-e,module_start)
 vita_create_self(${DM_APP_NAME}.suprx ${DM_APP_NAME} UNSAFE MODULE_ENTRY "module_start,module_stop,module_exit")
endif()
target_include_directories(${DM_APP_NAME} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" ${DM_APP_INCLUDE_DIRS})
target_compile_definitions(${DM_APP_NAME} PRIVATE DM_TRACK_MEMORY)
target_link_options(${DM_APP_NAME} PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc -Wl,--wrap=free)
target_compile_options(${DM_APP_NAME} PRIVATE -Wall -Wextra -O2)

find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(DM_APP_LINUX)
 set(DM_PAYLOAD "$<TARGET_FILE:${DM_APP_NAME}>")
 set(DM_PLATFORM linux)
 set(DM_PACKAGE_DEPENDENCY ${DM_APP_NAME})
else()
 set(DM_PAYLOAD "${CMAKE_CURRENT_BINARY_DIR}/${DM_APP_NAME}.suprx")
 set(DM_PLATFORM vita)
 set(DM_PACKAGE_DEPENDENCY ${DM_APP_NAME}.suprx-self)
endif()
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${DM_APP_NAME}.dmapp"
 COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/package-app.py ${DM_PAYLOAD} ${DM_PLATFORM} "${CMAKE_CURRENT_BINARY_DIR}/${DM_APP_NAME}.dmapp"
 DEPENDS ${DM_PACKAGE_DEPENDENCY} ${DM_PAYLOAD} ${CMAKE_CURRENT_SOURCE_DIR}/package-app.py)
add_custom_target(${DM_APP_NAME}-package ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/${DM_APP_NAME}.dmapp")
